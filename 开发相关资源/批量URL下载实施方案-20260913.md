## 1. 问题本质

当前 App 是"一条 URL → 一篇文章 → 选图 → 批量下载"。新需求是"一组 URL → 一篇文章接一篇文章 → 自动依次完成"。

本质差异是**外层队列**:从单文章管线升级为多文章串行调度。每次循环:

```
for url in urls:
  parseArticle(url)        ← 已有,串行即可
  ↓
  images = result.article.images
  ↓
  runDownload(article)     ← 需要新增的 Promise 化入口
  ↓
  onQueueFinished → resolve
```

## 5. 状态机(每条 URL)

```
pending → parsing → downloading → done
                              → failed
```

| 状态 | 显示 | 备注 |
|---|---|---|
| `pending` | `⏸ 等待` | 列表默认 |
| `parsing` | `🔄 解析中…` | `onStage` 推进 |
| `downloading` | `⏳ 下载中 18/48` | 实时进度 |
| `done` | `✅ 完成 24 张` | 终态 |
| `failed` | `❌ 0 张 / 失败` | 终态,带原因 |
| `skipped` | `⏭ 跳过` | 用户主动跳 |

## 6. 调度骨架

```ts
// src/screens/BatchListScreen.tsx
const runBatch = useCallback(async (urls: string[]) => {
  for (const url of urls) {
    if (cancelledRef.current) break;
    if (skippedRef.current.has(url)) continue;
    if (pausedRef.current) { await waitUntilResumed(); }

    setStatus(url, 'parsing');
    let result: ParseResult;
    try {
      result = await parseArticle(url, { signal: batchAbort.signal });
    } catch {
      setStatus(url, 'failed', '解析异常');
      continue;
    }
    if (!result.ok || result.article.images.length === 0) {
      setStatus(url, 'failed', result.error?.message ?? '无图');
      continue;
    }

    setStatus(url, 'downloading');
    try {
      await downloadContext.runDownload(result.article, {
        onProgress: (cur, total) =>
          setStatus(url, 'downloading', `下载中 ${cur}/${total}`),
      });
      setStatus(url, 'done', `${result.article.images.length} 张`);
    } catch {
      setStatus(url, 'failed', '下载失败');
    }
  }
  showSummary(); // Alert: 成功 N / 失败 M / 总 P
}, [downloadContext]);
```

## 7. 关键新增契约:`DownloadContext.runDownload`

把 `DownloadContext.start` 改造为返回 `Promise<void>`,在队列彻底清空时调用 `opts.onComplete`:

```ts
// src/store/DownloadContext.tsx
const runDownload = useCallback(
  async (
    article: TelegraphArticle,
    opts?: { onComplete?: () => void; onProgress?: (cur: number, total: number) => void }
  ) => {
    return new Promise<void>((resolve, reject) => {
      if (!queueRef.current) { reject(new Error('queue not ready')); return; }
      const settings = getSettingsSync();
      const relativePath = computeRelativePath(article, settings);
      const subfolderLeaf = computeSubfolder(article, settings);
      const initState = initDownloadState(article, article.images, relativePath, subfolderLeaf);
      let qState: DownloadState = initState;
      const q = createQueue({
        getState: () => qState,
        dispatch: a => { qState = downloadReducer(qState, a); reactDispatch(a); },
        onQueueFinished: () => { opts?.onComplete?.(); resolve(); },
      });
      queueRef.current.cancel();
      queueRef.current = q;
      reactDispatch({
        type: 'init',
        article,
        images: article.images,
        subfolder: relativePath,
        subfolderLeaf,
      });
      q.start();
    });
  },
  [],
);
```

### `downloadQueue` 改造

在 `drain` 退出循环(`live === 0 && pending.size === 0`)时调用 `onQueueFinished?.()`:

```ts
// src/store/downloadQueue.ts drain() 末尾
if (live === 0 && pending.size === 0) {
  console.log('[DL] drain end, queue finished');
  onQueueFinished?.();
}
```

`createQueue` 的 options 接受 `onQueueFinished?: () => void`;其它逻辑不动。

## 9. 取消 / 暂停 / 跳过的实现

| 操作 | 实现 |
|---|---|
| **取消全部** | `batchAbort.abort()` + `queueRef.current.cancel()` + 跳出 `for` 循环;当前条状态记 `cancelled`,其余保持 `pending` |
| **暂停** | 当前条跑完后 `pausedRef.current = true`;`for` 下一轮前 `await waitUntilResumed()`;用户点开始 → `pausedRef.current = false` |
| **跳过当前** | 把当前 url 加入 `skippedRef` + `queueRef.current.cancel()`;`for` 自然推进 |

## 10. 原生:`pickTextFile`(仿 `pickSaveDirectory`)

### Kotlin 端

```kotlin
// android/app/src/main/java/com/acstd/tgimagesdownloader/downloader/TelegraphDownloaderModule.kt

private var pickTextLauncher: Promise? = null

@ReactMethod
fun pickTextFile(promise: Promise) {
  try {
    val activity = currentActivity
    if (activity == null) { promise.resolve(null); return }
    val intent = Intent(Intent.ACTION_OPEN_DOCUMENT).apply {
      addCategory(Intent.CATEGORY_OPENABLE)
      type = "text/plain"
      // 若要支持更多类型:
      // type = "*/*"; putExtra(Intent.EXTRA_MIME_TYPES, arrayOf("text/plain", "text/*"))
    }
    pickTextLauncher = promise
    activity.startActivityForResult(intent, REQ_PICK_TEXT)
  } catch (e: Throwable) {
    pickTextLauncher = null
    promise.reject(ERR_UNKNOWN, e.message ?: "pickTextFile failed", e)
  }
}

private fun readPickedText(uri: Uri): String {
  val cr = reactApplicationContext.contentResolver
  return cr.openInputStream(uri)?.use { input ->
    input.readBytes().toString(Charsets.UTF_8).trimStart('\uFEFF')
  } ?: ""
}
```

### `MainActivity.kt` 的 `onActivityResult`

仿 `REQ_PICK_SAVE` 的处理,加一个 `REQ_PICK_TEXT` 分支:

```kotlin
private const val REQ_PICK_TEXT = 4322  // 与 REQ_PICK_SAVE 不冲突

override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
  super.onActivityResult(requestCode, resultCode, data)
  if (requestCode == REQ_PICK_SAVE && module != null) { /* existing */ }
  else if (requestCode == REQ_PICK_TEXT) {
    val module = reactInstanceManager
      .currentReactContext
      ?.getNativeModule(TelegraphDownloaderModule::class.java)
    val promise = module?.consumePickTextPromise()
    if (promise != null && resultCode == Activity.RESULT_OK && data?.data != null) {
      try {
        val uri = data.data!!
        val cr = contentResolver
        val name = cr.query(uri, arrayOf(android.provider.OpenableColumns.DISPLAY_NAME), null, null, null)
          ?.use { if (it.moveToFirst()) it.getString(0) else null }
          ?: "urls.txt"
        val content = module.readPickedText(uri)  // 公开只读方法
        val map = WritableNativeMap().apply {
          putString("uri", uri.toString())
          putString("name", name)
          putString("content", content)
        }
        promise.resolve(map)
      } catch (e: Throwable) {
        promise.reject("ERR_PICK_TEXT", e.message, e)
      }
    } else {
      promise?.resolve(null)
    }
  }
}
```

需要在 `TelegraphDownloaderModule.kt` 里加 `fun consumePickTextPromise(): Promise?` 和 `fun readPickedText(uri: Uri): String` 两个公开方法供 Activity 调用。

## 12. 单测要点

- **去重**: `[a, b, a, c, "", " "]` → `[a, b, c]`。
- **跳过继续**: 3 条 URL,第 2 条标记 `skipped` → 第 3 条仍执行,最终 `done=2, skipped=1`。
- **失败跳过**: 第 2 条 `parseArticle` 返回 `ok:false` → 第 3 条仍执行,`done=2, failed=1`。
- **取消**: `batchAbort.abort()` → `for` 跳出,已下完的留 `done`,未开始的保持 `pending`。
- **暂停恢复**: `pausedRef.current = true` 时循环挂起;`false` 后继续。
- **`runDownload` 完成回调**: mock 队列 `drain` 触发 `onQueueFinished` → Promise resolve,状态转 `done`。

## 13. 风险与对策

| 风险 | 对策 |
|---|---|
| 100 条串行慢 | MVP 接受;后续在 `runBatch` 外层加"解析并行 + 下载串行" |
| 后台被杀 | 已有前台服务通知(`FOREGROUND_SERVICE_DATA_SYNC`),长任务提醒保持前台 |
| `url.txt` BOM / 编码异常 | 读取时 `trimStart('\uFEFF')`,UTF-8 兜底;空行 `filter(Boolean)` |
| 现有单 URL 流程回归 | `runDownload` 不替换原 `start`;PreviewScreen 仍走 `start` |
| 批量中用户按 Home | 切后台队列仍跑(前台服务保活);回前台 `useFocusEffect` 从 React 状态读取已完成的项 |
| `parseArticle` 的 AbortSignal | 批量"取消全部"要级联到当前正在解析的 URL — `parseArticle({signal})` 已有,直接传 `batchAbort.signal` |
